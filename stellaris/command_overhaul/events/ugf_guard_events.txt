namespace = ugf_guard

# Capital Guards: 2 at Planetary Administration tier, 4 at Major Capital tier,
# 6 at System / Imperial Capital tier (habitat equivalents included).
# Fired every month from on_monthly_pulse_country (this = country), see
# common/on_actions/ugf_on_actions.txt.
# Every owned colony with a capital building is topped up to its tier's target,
# counting the ugf_capital_guard armies already standing on the planet, so the
# event is safe to fire repeatedly, needs no planet flags, works on existing
# saves, and replaces guards lost in ground combat.
country_event = {
	id = ugf_guard.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		is_gestalt = no
	}
	immediate = {
		every_owned_planet = {
			limit = {
				has_owner = yes
				has_ground_combat = no
				OR = {
					has_building = building_capital
					has_building = building_hab_capital
					has_building = building_major_capital
					has_building = building_hab_major_capital
					has_building = building_system_capital
					has_building = building_imperial_capital
					has_building = building_hab_system_capital
				}
			}
			# Tier 3: System / Imperial capital -> 6 guards
			if = {
				limit = {
					OR = {
						has_building = building_system_capital
						has_building = building_imperial_capital
						has_building = building_hab_system_capital
					}
				}
				while = {
					count = 6
					if = {
						limit = {
							count_planet_army = {
								limit = { army_type = ugf_capital_guard }
								count < 6
							}
						}
						create_army = {
							name = "ugf_capital_guard_army_name"
							owner = root
							species = root.species
							type = ugf_capital_guard
						}
					}
				}
			}
			# Tier 2: Major capital -> 4 guards
			else_if = {
				limit = {
					OR = {
						has_building = building_major_capital
						has_building = building_hab_major_capital
					}
				}
				while = {
					count = 4
					if = {
						limit = {
							count_planet_army = {
								limit = { army_type = ugf_capital_guard }
								count < 4
							}
						}
						create_army = {
							name = "ugf_capital_guard_army_name"
							owner = root
							species = root.species
							type = ugf_capital_guard
						}
					}
				}
			}
			# Tier 1: Planetary Administration -> 2 guards
			else = {
				while = {
					count = 2
					if = {
						limit = {
							count_planet_army = {
								limit = { army_type = ugf_capital_guard }
								count < 2
							}
						}
						create_army = {
							name = "ugf_capital_guard_army_name"
							owner = root
							species = root.species
							type = ugf_capital_guard
						}
					}
				}
			}
		}
	}
}
